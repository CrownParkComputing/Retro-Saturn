cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(RetroSaturnCore LANGUAGES C CXX)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(YMIR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../core")
set(Ymir_LIBRARY_ONLY ON CACHE BOOL "" FORCE)
set(Ymir_ENABLE_IPO OFF CACHE BOOL "" FORCE)
set(Ymir_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(Ymir_ENABLE_SANDBOX OFF CACHE BOOL "" FORCE)
set(Ymir_ENABLE_YMDASM OFF CACHE BOOL "" FORCE)
set(Ymir_DEV_BUILD OFF CACHE BOOL "" FORCE)
set(Ymir_EXTRA_INLINING OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_Vulkan ON CACHE BOOL "" FORCE)
find_package(Threads REQUIRED)
if(APPLE)
  # The app embeds the software renderer. Keep upstream C++ untouched while
  # excluding its standalone GPU shader-tool dependency from this build.
  set(YMIR_ORIGINAL_ROOT "${YMIR_ROOT}")
  set(YMIR_ROOT "${CMAKE_CURRENT_BINARY_DIR}/ymir-software-source")
  file(MAKE_DIRECTORY "${YMIR_ROOT}")
  file(GLOB ymir_entries LIST_DIRECTORIES true "${YMIR_ORIGINAL_ROOT}/*")
  foreach(entry IN LISTS ymir_entries)
    get_filename_component(name "${entry}" NAME)
    if(NOT name STREQUAL "CMakeLists.txt" AND NOT EXISTS "${YMIR_ROOT}/${name}")
      file(CREATE_LINK "${entry}" "${YMIR_ROOT}/${name}" SYMBOLIC)
    endif()
  endforeach()
  file(READ "${YMIR_ORIGINAL_ROOT}/CMakeLists.txt" ymir_cmake)
  string(REPLACE "include(cmake/CompileShaders.cmake)"
    "include(\"${CMAKE_CURRENT_LIST_DIR}/cmake/SoftwareShaders.cmake\")" ymir_cmake "${ymir_cmake}")
  file(CONFIGURE OUTPUT "${YMIR_ROOT}/CMakeLists.txt" CONTENT "${ymir_cmake}" @ONLY)
endif()
add_subdirectory("${YMIR_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/ymir")

set(BRIDGE_SOURCES bridge/ymir_bridge.cpp bridge/savestate_io.cpp bridge/audio_backend_stub.cpp)
if(ANDROID)
  list(APPEND BRIDGE_SOURCES bridge/audio_backend_android.cpp)
elseif(APPLE)
  list(APPEND BRIDGE_SOURCES bridge/audio_backend_sdl.cpp)
else()
  list(APPEND BRIDGE_SOURCES bridge/audio_backend.cpp)
endif()
set(BRIDGE_KIND SHARED)
if(APPLE)
  set(BRIDGE_KIND STATIC)
endif()
add_library(ymircore ${BRIDGE_KIND} ${BRIDGE_SOURCES})
target_compile_features(ymircore PUBLIC cxx_std_20)
target_include_directories(ymircore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/bridge")
target_link_libraries(ymircore PUBLIC ymir-core Threads::Threads)
if(ANDROID)
  target_link_libraries(ymircore PRIVATE aaudio android log)
elseif(APPLE)
  target_link_libraries(ymircore PUBLIC SDL3::SDL3)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(ALSA REQUIRED IMPORTED_TARGET alsa)
  target_link_libraries(ymircore PRIVATE PkgConfig::ALSA)
endif()
set_target_properties(ymircore PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

option(RETRO_SATURN_CORE_TESTS "Build the native bridge regression checks" OFF)
if(RETRO_SATURN_CORE_TESTS AND NOT ANDROID AND NOT APPLE)
  enable_testing()
  foreach(name smoke peripheral swap_snapshot audio_tone input_mixdown backup_ram savestate pause core_options)
    add_executable(ymir_${name}_test "linux/test/ymir_${name}_test.cpp")
    target_link_libraries(ymir_${name}_test PRIVATE ymircore)
    set_target_properties(ymir_${name}_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    add_test(NAME ymir_${name} COMMAND ymir_${name}_test)
  endforeach()
endif()
